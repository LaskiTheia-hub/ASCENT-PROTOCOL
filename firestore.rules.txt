rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function isAdmin() { return signedIn() && request.auth.token.email in ["operator@argus.test"]; }

    match /progress/{uid} {
      allow read: if isAdmin() || (signedIn() && uid == request.auth.uid);
      allow create, update: if signedIn() && (uid == request.auth.uid || isAdmin());
    }

    match /events/{eventId} {
      allow read: if isAdmin();
      allow create: if signedIn() && request.resource.data.uid == request.auth.uid;
    }
  }
}
