rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() {
      return request.auth != null;
    }
    function email() {
      return request.auth.token.email;
    }
    function userId() {
      return request.auth.token.email.split("@")[0];
    }
    function isAdmin() {
      return signedIn() && email() in ["operator@argus.test"];
    }

    match /progress/{id} {
      allow read: if isAdmin() || (signedIn() && id == userId());
      allow create, update: if signedIn() && (id == userId() || isAdmin());
    }

    match /events/{eventId} {
      allow read: if isAdmin();
      allow create: if signedIn();
    }
  }
}
